schema for uctp_output.json

This is the expected form of the output of the UCT processor, which is an input to the evaluator
This schema coincides with the schema used by industry partners currently publishing UCT solutions
NOTE: None of the values listed below "cov" are currently used in the evaluator

[
  {
    "idStateVector":         unique identifier assigned to this object, value is arbitrary,
    "sourcedData":           [list of oberavtion "id"s correlated to this object],
    "sourcedDataTypes":      [list of the type of data each observation is, currently on supported datatype is "EO"],
    "classificationMarking": unused parameter
    "epoch":                 time at which the following state vector is valid
    "xpos":  x-position of satellite [km],
    "ypos":  y-'''            ''',
    "zpos":  z-'''            ''',
    "xvel":  x-velocity of satellite [km/s],
    "yvel":  y-'''            ''',
    "zvel":  z-'''            ''', 
    "referenceFrame":    reference frame of above state vector components, assume J2000 if Null
    "covReferenceFrame": reference frame of covariance matrix
    "cov":               [list of 21 upper diagonal elements of symmetric covariance matrix],
    "lunarSolar":      solution obtained using 3rd body gravity perturbations (true/false),
    "solarRadPress":   solution obtrained using solar radiation pressure perturbations (true/false),
    "solidEarthTides": solution obtained using earth tides perturbations (true/false),
    "inTrackThrust":   solution obtained accounting for in-track thrust of object (true/false),
    "rms": 	       root mean square error of residuals, this value is recalculated as a metric in the evaluator
    "source":          data source,
    "dataMode":        type of data ("REAL/SIMULATED"),
    "algorithm":       algorithm used to obtain solution
  },
  {
    ...
    ...
  }
}
